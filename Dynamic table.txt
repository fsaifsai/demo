create schema dynamic_tables_schema;

CREATE TABLE movies (
movie_id INT PRIMARY KEY,
title VARCHAR(100),
release_year INT,
genre VARCHAR(50)
);


CREATE TABLE movie_ratings (
rating_id INT PRIMARY KEY,
movie_id INT,
user_id INT,
rating DECIMAL(3,1),
rating_date DATE,
FOREIGN KEY (movie_id) REFERENCES movies(movie_id)
);

create or replace dynamic table movie_ratings_dt
target_lag='1 minutes'
warehouse='PRACTICE'
as select m.movie_id,m.title,avg(r.rating) as average_rating
from movies m join movie_ratings r
on m.movie_id=r.movie_id
group by m.movie_id,m.title;

select * from movie_ratings_dt;

target_lag=downstream
warehouse='PRACTICE'
as select title,average_rating from movie_ratings_dt order by average_rating desc limit 2;
